# --- Test harness (backend/test_phase1_e2e.py) ------------------------------ #
# The Phase-1 E2E suite compares rendered frames pixel by pixel, so it imports
# Pillow and NumPy directly. NumPy is also pulled in transitively by
# onnxruntime (via faster-whisper), but because it is imported directly it has
# to be declared here - otherwise the suite fails with ModuleNotFoundError on
# a clean CI runner.
pillow==13.0.0
numpy==2.5.3

# --- Local AI (optional GPU acceleration) ---------------------------------- #
# Optional: cloud-only setups can skip this. GPU transcription is optional -
# the CPU fallback works out of the box; the app installs the CUDA runtime pack
# separately and passes its directory as TAFSEER_CUDA_DIR.
faster-whisper>=1.1,<2
# PyAV is a faster-whisper dependency, but it must be pinned: faster-whisper
# 1.2.x calls `av.open(..., metadata_errors="ignore")`, which PyAV 17+ removed
# (TypeError at the first real transcription). 16.1.0 is the newest release that
# still accepts the argument.
av==16.1.0
